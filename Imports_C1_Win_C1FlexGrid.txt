    ''DENSO WAVE Active USB-COM Port 取得
    Private Sub Init_DENSO_USB_COM_Port()

        Dim mcW32SerPort As New ManagementClass("Win32_SerialPort")
        Dim aSerialPort As ManagementObject
        Dim dCOM As String

        For Each aSerialPort In mcW32SerPort.GetInstances()
            Console.WriteLine(DirectCast(aSerialPort.GetPropertyValue("DeviceID"), String) & ":" & DirectCast(aSerialPort.GetPropertyValue("Caption"), String))
            'DENSO WAVE Active USB-COM Port 検索 AND Conneced検索
            dCOM = DirectCast(aSerialPort.GetPropertyValue("DeviceID"), String) & ":" & DirectCast(aSerialPort.GetPropertyValue("Caption"), String)

            If 0 <= dCOM.IndexOf("DENSO WAVE Active USB-COM Port") Then
                If 0 <= dCOM.IndexOf("Connected") Then
                    '接続ポート
                    CMP = Strings.Left(dCOM, 4)
                    Exit Sub
                Else

                End If
            Else
            End If
        Next
        CMP = ""

    End Sub
    Private Sub COM_Connect()

        Try
            If CMP <> "" Then
                '19200,N,8,1 通信設定
                With SerialPort1
                    .PortName = CMP
                    .BaudRate = 19200
                    .DataBits = 8
                    .Parity = Parity.None
                    .StopBits = StopBits.One
                    .Handshake = Handshake.None
                    .ReceivedBytesThreshold = 2
                    .ReadTimeout = 10000
                    .WriteTimeout = 10000

                    .DtrEnable = True
                    .RtsEnable = True
                End With

                If SerialPort1.IsOpen = False Then
                    SerialPort1.Open()
                    SerialPort1.DiscardInBuffer()
                    SerialPort1.DiscardOutBuffer()
                End If
            Else
                MessageBox.Show(" QRコードリーダーの接続が確認できません　", "警告", MessageBoxButtons.OK, MessageBoxIcon.Error)
            End If

        Catch ex As Exception
            Call ErrLOG("COM_Connect", "")
            MessageBox.Show(" ハンディターミナルが接続されていません??　", "ERROR", MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try

    End Sub
    Private Sub COM_DisConnect()
        If SerialPort1.IsOpen = True Then
            SerialPort1.Close()
        End If
    End Sub

    'バーコードリーダーからの入力
    Private Sub SerialPort1_DataReceived(sender As Object, e As System.IO.Ports.SerialDataReceivedEventArgs) Handles SerialPort1.DataReceived

        If SerialPort1.IsOpen = False Then
            Return
        End If
        System.Threading.Thread.Sleep(200)
        'Clear
        blockbuf = ""
        Do
            Dim returnStr As String = SerialPort1.ReadExisting()
            blockbuf &= returnStr.ToString

            If SerialPort1.BytesToRead = 0 Then
                SerialPort1.DiscardInBuffer()
                SerialPort1.DiscardOutBuffer()
                Exit Do
            End If
        Loop

        '表示対応
        If Strings.Left(blockbuf, 2) = "DK" Then






        End If

    End Sub

    '作業ポイントDB登録 座標+作業番号+作業種類
    Private Sub PointDBIN(Xn As Integer, Yn As Integer, WN As String, WT As Integer)

        Dim cma As OleDb.OleDbCommand
        Dim strSQL As String

        If cna.State = ConnectionState.Open = True Then
            cna.Close()
        End If

        If WT <> 99 And WT <> 98 Then
            'accdb INSERT
            strSQL = ""
            strSQL = strSQL + "INSERT INTO work_pointdata("
            strSQL = strSQL + "workno,"
            strSQL = strSQL + "wtime,"
            strSQL = strSQL + "x,"
            strSQL = strSQL + "y,"
            strSQL = strSQL + "wtype,"
            strSQL = strSQL + "sflg"
            strSQL = strSQL + ")VALUES("
            strSQL = strSQL + "'" & WN & "',"
            strSQL = strSQL + "'" & Now() & "',"
            strSQL = strSQL + "" & Xn & ","
            strSQL = strSQL + "" & Yn & ","
            strSQL = strSQL + "" & WT & ","
            strSQL = strSQL + "'" & "0" & "'"
            strSQL = strSQL + ")"
            cna.Open()
            cma = New OleDb.OleDbCommand
            cma.Connection = cna
            cma.CommandText = strSQL
            cma.ExecuteNonQuery()
            cna.Close()

        ElseIf WT = 98 Then
            '作業履歴クリアの場合
            strSQL = ""
            strSQL = strSQL + "DELETE FROM work_pointdata "
            strSQL = strSQL + "WHERE workno='" & WN & "' "
            strSQL = strSQL + "AND x=" & Xn & " "
            strSQL = strSQL + "AND y=" & Yn & " "
            cna.Open()
            cma = New OleDb.OleDbCommand
            cma.Connection = cna
            cma.CommandText = strSQL
            cma.ExecuteNonQuery()
            cna.Close()

        End If

    End Sub

    '作業履歴をCSV出力
    Private Sub OutputCSV(WN As String)

        Dim cma As OleDb.OleDbCommand
        Dim strSQL As String
        Dim rsa As OleDb.OleDbDataReader
        Dim Num As Integer


        Dim FileName As String
        Dim WS As System.IO.StreamWriter
        Dim strTXT As String


        If cna.State = ConnectionState.Open = True Then
            cna.Close()
        End If

        '念のため
        strSQL = ""
        strSQL = strSQL + "SELECT COUNT(*) FROM work_pointdata "
        strSQL = strSQL + "WHERE workno='" & WN & "' "
        cna.Open()
        cma = New OleDb.OleDbCommand
        cma.Connection = cna
        cma.CommandText = strSQL
        Num = cma.ExecuteScalar
        cna.Close()

        If Num <> 0 Then

            'CSV作成準備
            FileName = System.Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory) & "\" & WN & ".csv"
            WS = System.IO.File.AppendText(FileName)

            strSQL = ""
            strSQL = strSQL + "SELECT * FROM work_pointdata "
            strSQL = strSQL + "WHERE workno='" & WN & "' "
            cna.Open()
            cma = New OleDb.OleDbCommand
            cma.Connection = cna
            cma.CommandText = strSQL
            rsa = cma.ExecuteReader
            While (rsa.Read())

                strTXT = ""
                strTXT = strTXT + rsa("workno").ToString & "," & rsa("wtime").ToString & "," & rsa("x").ToString & "," & rsa("y").ToString & "," & rsa("wtype").ToString
                WS.WriteLine(strTXT)

            End While
            cna.Close()
            WS.Close()

        End If

    End Sub
