    'INI.xml
    '必要変数取得 INI.XML
    Private Sub Initialization()
        Try
            'INI.XMLから設定取得
            Dim MyPath As String = My.Application.Info.DirectoryPath
            Dim iniFileName As String = "INI.xml"
            If MyPath.EndsWith("\") = False Then
                MyPath &= "\"
            End If
            iniFileName = MyPath & iniFileName

            '指定したXMLファイルの読み込み
            Dim xmlDoc As New System.Xml.XmlDocument
            xmlDoc.Load(iniFileName)
            Dim rootElement As System.Xml.XmlElement = xmlDoc.DocumentElement
            Dim node As System.Xml.XmlNodeList = rootElement.GetElementsByTagName("db_ip")

            '指定したタグが存在するか？
            If node.Count > 0 Then
                DBIP = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_port")
            If node.Count > 0 Then
                DBPO = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_name")
            If node.Count > 0 Then
                DBNM = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_backf")
            If node.Count > 0 Then
                TForder = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("app_path")
            If node.Count > 0 Then
                APPTH = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("s_date")
            If node.Count > 0 Then
                SD = node.Item(0).InnerText
            End If

            '熱処理進捗
            node = rootElement.GetElementsByTagName("dbh_ip")
            If node.Count > 0 Then
                DHIP = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("dbh_port")
            If node.Count > 0 Then
                DHPO = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("dbh_name")
            If node.Count > 0 Then
                DHNM = node.Item(0).InnerText
            End If

        Catch ex As Exception
            Call ErrLOG("Initialization", "")
        End Try
    End Sub
    'エラーログを書き出す
    Public Sub ErrLOG(ByVal C1 As String, ByVal C2 As String)
        Dim FileName As String
        Dim WS As IO.StreamWriter

        Try
            FileName = My.Application.Info.DirectoryPath & "\Errlog.txt"
            WS = IO.File.AppendText(FileName)
            WS.WriteLine(C1 & ":" & C2 & ":" & Format(Now, "yyyy/MM/dd HH:mm:ss"))
            WS.Close()
        Catch ex As Exception
            End
        End Try
    End Sub

    '進捗確認用
    Public Sub ProcessLOG(ByVal C1 As String, ByVal C2 As String)
        Dim FileName As String
        Dim WS As IO.StreamWriter

        Try
            FileName = My.Application.Info.DirectoryPath & "\proc.txt"
            WS = IO.File.AppendText(FileName)
            WS.WriteLine(C1 & ":" & C2 & ":" & Format(Now, "yyyy/MM/dd HH:mm:ss"))
            WS.Close()
        Catch ex As Exception
            End
        End Try
    End Sub

    'CSV取得1
    Private Sub DataMake01()

        Dim cm As NpgsqlCommand
        Dim strSQL As String
        Dim Num As Integer
        Dim rs As NpgsqlDataReader

        Dim SDate As String

        Try

            If cn1.State = ConnectionState.Open = True Then
                cn1.Close()
            End If

            'init_folder workno=1 有効のみ 対象フォルダ検索
            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM init_folder "
            strSQL = strSQL + "WHERE workno='1' AND vif='0' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num = cm.ExecuteScalar
            cn1.Close()

            If Num = 0 Then
                '有効フォルダなし
                prflg = False
                Exit Sub

            Else
                '取得対象日設定
                SDate = DateAdd(DateInterval.Day, SD * (-1), Now())
                prflg = True

            End If

            strSQL = ""
            strSQL = strSQL + "SELECT * FROM init_folder "
            strSQL = strSQL + "WHERE workno='1' AND vif='0' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            rs = cm.ExecuteReader
            While (rs.Read())

                '進捗確認用
                If procsv = True Then
                    Call ProcessLOG("DataMake01", rs("loc_folder"))
                End If

                'BAT処理 CSV取込
                Call DataMake02K(rs("loc_folder"), SDate)

            End While
            cn1.Close()

        Catch ex As Exception
            Call ErrLOG("DataMake01", "")
        End Try

    End Sub

    'SEDコマンドを使用しないで先頭3行削除作業
    Private Sub DataMake02K(FOL As String, SDate As String)

        Dim KDate As String                             'ファイル更新日時
        Dim StrF As String = String.Empty
        Dim sr As System.IO.StreamReader = Nothing
        Dim sw As System.IO.StreamWriter = Nothing

        Dim jCnt As Integer = 0

        'Shift-JISエンコードで変換できない場合は「?」文字の設定'
        Dim encFallBack As System.Text.DecoderReplacementFallback = New System.Text.DecoderReplacementFallback("?")
        Dim enc As System.Text.Encoding = System.Text.Encoding.GetEncoding("shift_jis", System.Text.EncoderFallback.ReplacementFallback, encFallBack)

        Dim r As New System.Random(12345)

        For Each FName In IO.Directory.GetFiles(FOL, "*.csv", SearchOption.AllDirectories)

            'ファイル更新日取得
            KDate = System.IO.File.GetLastWriteTime(FName)

            If KDate > SDate Then

                Dim rm As Integer = r.Next(99999999)


                'ファイル名のみから別名設定
                StrF = T5 & "\" & CStr(rm) & "_" & System.IO.Path.GetFileName(FName)
                'StrF = T5 & "\" & Format(Now, "HHmmss") & "_" & System.IO.Path.GetFileName(FName)

                'originalファイルを1-3行削除
                sr = New System.IO.StreamReader(FName, enc)

                '一時ファイルを作成する
                Dim tmpPath As String = System.IO.Path.GetTempFileName()
                '一時ファイルを書き込みで開く
                sw = New System.IO.StreamWriter(StrF, True, enc)

                '先頭フラグON 
                Dim blnFirstLine As Boolean = True
                'テキストを一行ずつ読込
                While sr.Peek() > -1
                    '一行読込
                    Dim line As String = sr.ReadLine()
                    '先頭行1-3行目スキップ
                    If blnFirstLine = True Then
                        '先頭フラグONの場合、OFFしてスキップ
                        If jCnt > 1 Then
                            blnFirstLine = False
                        End If

                        jCnt = jCnt + 1
                        Continue While
                    End If
                    '一時ファイル書込
                    sw.WriteLine(line)
                End While
                '閉じる
                sr.Close()
                sw.Close()
                sr = Nothing
                sw = Nothing

                jCnt = 0

                'temp削除
                System.IO.File.Delete(tmpPath)
                '元ファイル削除 削除する必要なし
                'IO.File.Delete(FName)

            End If
        Next

    End Sub

    'CSV取り込み
    Private Sub DB_Import01()

        Dim proc As New Process
        Dim ImportFileName As String

        Dim i As Integer
        Dim FN() As String
        Dim dft As DateTime = DateTime.Parse(Now)
        Dim j As Integer

        Dim l As Long

        Try
            'フォルダ内ファイル有無確認
            i = 0
            For Each strFilePath In System.IO.Directory.GetFiles(T5, "*.csv")
                ReDim Preserve FN(i)

                FN(i) = System.IO.Path.GetFileName(strFilePath)
                i = i + 1
                '    End If
            Next

            '対象ファイル処理
            For j = 0 To i - 1

                'ファイルサイズチェック
                'FileInfo の新しいインスタンスを生成する 
                Dim fi As New System.IO.FileInfo(T5 & "\" & FN(j))
                'ファイルのサイズを取得 
                l = fi.Length
                '0KB処理しない
                If l <> 0 Then

                    ImportFileName = T5 & "\" & FN(j)
                    '処理用のファイル名で処理 
                    proc.StartInfo.FileName = APPTH & "\psql.exe"
                    proc.StartInfo.Arguments = " -h " & DBIP & " -p 25432 -U postgres -d " & DBNM & " -c " & Chr(34) & "\copy trace5_temp FROM " & Chr(39) & ImportFileName & Chr(39) & " with csv ;"
                    'proc.StartInfo.Arguments = " -h 192.168.23.147 -p 25432 -U postgres -d vmdb -c " & Chr(34) & "\copy dataupdate FROM " & Chr(39) & ImportFileName & Chr(39) & " with csv;"
                    proc.StartInfo.UseShellExecute = False
                    proc.StartInfo.CreateNoWindow = True
                    proc.Start()
                    proc.WaitForExit()
                    proc.Close()
                    'proc.Dispose()
                    'proc = Nothing
                    System.Threading.Thread.Sleep(50)

                End If

            Next

            '取込済みファイル削除 
            For j = 0 To i - 1

                '進捗確認用
                If procsv = True Then
                    Call ProcessLOG("DataImport01", T5 & "\" & FN(j))
                End If
                'TEMP_CSV
                IO.File.Delete(T5 & "\" & FN(j))

            Next

        Catch ex As Exception
            Call ErrLOG("DB_Import01", "")
        End Try

    End Sub
