    '必要変数取得 INI.XML
    Private Sub Initialization()
        Try
            'INI.XMLから設定取得
            Dim MyPath As String = Application.StartupPath
            Dim iniFileName As String = "INI.xml"
            If MyPath.EndsWith("\") = False Then
                MyPath &= "\"
            End If
            iniFileName = MyPath & iniFileName

            '指定したXMLファイルの読み込み
            Dim xmlDoc As New System.Xml.XmlDocument
            xmlDoc.Load(iniFileName)
            Dim rootElement As System.Xml.XmlElement = xmlDoc.DocumentElement
            Dim node As System.Xml.XmlNodeList = rootElement.GetElementsByTagName("db_name")

            '指定したタグが存在するか？
            If node.Count > 0 Then
                DBN = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_port")
            If node.Count > 0 Then
                DBP = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_ip")
            If node.Count > 0 Then
                DIP = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("usb_port")
            If node.Count > 0 Then
                CMP = node.Item(0).InnerText
            End If

            '-- 金型情報 --
            node = rootElement.GetElementsByTagName("db_port2")
            If node.Count > 0 Then
                DBP2 = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_ip2")
            If node.Count > 0 Then
                DIP2 = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_name2")
            If node.Count > 0 Then
                DBN2 = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("kmng")
            If node.Count > 0 Then
                KMNG = node.Item(0).InnerText
            End If


        Catch ex As Exception
            Call ErrLOG("Initialization", "")
            End
        End Try
    End Sub

    'エラーログを書き出す
    Public Sub ErrLOG(ByVal C1 As String, ByVal C2 As String)

        Dim FileName As String
        Dim WS As System.IO.StreamWriter

        Try
            FileName = PTH & "\errlog.txt"

            WS = System.IO.File.AppendText(FileName)
            WS.WriteLine(C1 & ":" & C2 & ":" & Format(Now, "yyyy/MM/dd HH:mm:ss"))
            WS.Close()
        Catch ex As Exception
            End
        End Try
    End Sub
    'COMポート確認
    Private Sub COM_Connect()

        Try
            If CMP <> "" Then
                '38400,N,8,1 通信設定
                With SerialPort1
                    .PortName = CMP
                    .BaudRate = 38400
                    .DataBits = 8
                    .Parity = Parity.None
                    .StopBits = StopBits.One
                    .Handshake = Handshake.None
                    .ReceivedBytesThreshold = 2
                    .ReadTimeout = 10000
                    .WriteTimeout = 10000

                    .DtrEnable = True
                    .RtsEnable = True
                End With

                If SerialPort1.IsOpen = False Then
                    SerialPort1.Open()
                    SerialPort1.DiscardInBuffer()
                    SerialPort1.DiscardOutBuffer()
                End If
            Else
                MessageBox.Show(" QRコードリーダーの接続が確認できません　", "警告", MessageBoxButtons.OK, MessageBoxIcon.Error)
            End If

        Catch ex As Exception
            'Call ErrLOG("COM_Connect", "")
            MessageBox.Show(" ハンディターミナルが接続されていません??　", "ERROR", MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try

    End Sub
    Private Sub COM_DisConnect()
        If SerialPort1.IsOpen = True Then
            SerialPort1.Close()
        End If

    Private Sub C1FlexGrid3_Click(sender As Object, e As EventArgs) Handles C1FlexGrid3.Click

        Dim C20 As String = ComboBox20.Text
        Dim y As Integer                        'select row

        Try
            y = C1FlexGrid3.Row
            If y = 0 Or y < 0 Then
                Exit Sub
            End If

            Select Case C20
                Case "工程"
                    '削除希望データ情報
                    D_Data = C1FlexGrid3(y, 4)
                Case "作業者"
                    D_Data = C1FlexGrid3(y, 1)
                Case "作業内容"
                    D_Data = C1FlexGrid3(y, 2)
                Case "90工程修理内容"
                    D_Data = C1FlexGrid3(y, 2)
                Case "異常内容"
                    D_Data = C1FlexGrid3(y, 2)
            End Select

            C1FlexGrid3.BeginUpdate()

            C1FlexGrid3.Styles.Add("White")
            C1FlexGrid3.Styles("White").BackColor = Color.White
            C1FlexGrid3.Styles.Add("red")
            C1FlexGrid3.Styles("red").BackColor = Color.Red

            If SR = y Then
                'リセット処理
                C1FlexGrid3.Rows(y).Style = C1FlexGrid3.Styles("White")
                D_Data = ""
                SR = 0

                TextBox3.Text = ""
                TextBox4.Text = ""

                Button7.Text = "新規登録"
                Button7.Enabled = True
                Button8.Visible = False
                Button8.Enabled = False
                C1FlexGrid3.Select(0, 0, True)

            Else
                'レコード選択
                C1FlexGrid3.Rows(y).Style = C1FlexGrid3.Styles("red")
                If SR > 0 Then
                    C1FlexGrid3.Rows(SR).Style = C1FlexGrid3.Styles("White")
                End If

                SR = y

                Select Case C20
                    Case "工程"
                        TextBox3.Text = C1FlexGrid3(y, 2)
                        TextBox4.Text = C1FlexGrid3(y, 3)
                    Case "作業者"
                        TextBox3.Text = C1FlexGrid3(y, 1)
                        TextBox4.Text = C1FlexGrid3(y, 2)
                    Case "作業内容", "90工程修理内容", "異常内容"
                        TextBox3.Text = C1FlexGrid3(y, 1)
                    Case Else

                End Select

                Button7.Text = "修正"
                Button7.Enabled = True
                Button8.Visible = True
                Button8.Enabled = True
                C1FlexGrid3.Select(0, 0, True)

            End If
            C1FlexGrid3.EndUpdate()

        Catch ex As Exception
            Call ErrLOG("C1FlexGrid3_Click", "")
        End Try

    End Sub

    Private Sub Button6_Click(sender As Object, e As EventArgs) Handles Button6.Click

        Dim C20 As String = ComboBox20.Text
        Dim C21 As String = ComboBox21.Text
        Dim T2 As String = TextBox2.Text
        Dim FN As String

        Dim Qdata As String

        Try
            If Len(T2) > 0 Then

                Select Case C20
                    Case "工程"
                        Qdata = "PR" & T2 & Format(Now, "yyyyMMdd")
                        FN = T2

                    Case "作業者"
                        Qdata = "JY" & T2 & Format(Now, "yyyyMMdd")
                        FN = T2

                    Case "作業内容", "90工程修理内容", "異常内容"
                        Qdata = "WK" & T2 & Format(Now, "yyyyMMdd")
                        FN = C21

                    Case "作業用ラベル"
                        Qdata = T2 & Format(Now, "yyyyMMdd")
                        FN = C21

                    Case Else

                End Select

                ComboBox21.Text = ""
                TextBox2.Text = ""

                ' QRCodeGenerator クラスを使って QRコードを生成
                Dim qrGenerator As New QRCodeGenerator()
#Disable Warning BC42104 ' 変数 'Qdata' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                Dim qrCodeData As QRCodeData = qrGenerator.CreateQrCode(Qdata, QRCodeGenerator.ECCLevel.Q)
#Enable Warning BC42104 ' 変数 'Qdata' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                Dim qrCode As New QRCode(qrCodeData)

                ' QRコードを Bitmap イメージとして生成
                Dim qrCodeImage As Bitmap = qrCode.GetGraphic(10)
                ' BMP 画像として保存
#Disable Warning BC42104 ' 変数 'FN' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                Dim outputPath As String = System.Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory) & "\" & FN & ".bmp"
#Enable Warning BC42104 ' 変数 'FN' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                qrCodeImage.Save(outputPath, ImageFormat.Bmp)

                'Console.WriteLine($"QRコードが生成され、{outputPath} に保存されました。")
                MessageBox.Show(" ラベルが発行されました ", "完了", MessageBoxButtons.OK, MessageBoxIcon.Information)

                Button6.Enabled = False

            End If

        Catch ex As Exception
            Call ErrLOG("Button6_Click", "")
        End Try

    End Sub

    Private Sub SerialPort1_DataReceived(sender As Object, e As System.IO.Ports.SerialDataReceivedEventArgs) Handles SerialPort1.DataReceived

        Dim WPP As String
        Dim WPN As String


        If SerialPort1.IsOpen = False Then
            Return
        End If
        System.Threading.Thread.Sleep(500)
        'Clear
        blockbuf = ""
        Do
            Dim returnStr As String = SerialPort1.ReadExisting()
            blockbuf &= returnStr.ToString

            If SerialPort1.BytesToRead = 0 Then
                SerialPort1.DiscardInBuffer()
                SerialPort1.DiscardOutBuffer()
                Exit Do
            End If
        Loop

        Select Case Strings.Left(blockbuf, 2)
            Case "DK"
                '金型
                DIEN = Strings.Mid(blockbuf, 3, 5)

                If Me.InvokeRequired Then
                    ComboBox5.Invoke(New dlgWriteTxt(AddressOf WriteTxt1), DIEN)
                Else
                    ComboBox5.Text = DIEN
                End If

            Case "JY"
                '従業員番号

                If Me.InvokeRequired Then
                    ComboBox3.Invoke(New dlgWriteTxt(AddressOf WriteTxt2), NtoName(EMPN))
                End If

            Case "PR"
                '工程用
                If PROC = "20" Then
                    KLNE = Strings.Mid(blockbuf, 4, 1)
                    KTER = Strings.Mid(blockbuf, 6, 2)

                    If Me.InvokeRequired Then
                        ComboBox4.Invoke(New dlgWriteTxt(AddressOf WriteTxt3), LKtoChange(KLNE, KTER))
                    End If

                End If

            Case "WP"

                If PROC = "30" Then

                        WPN = "通常作業開始"
                    End If

                        WPN = "作業終了"
                    End If

    '従業員番号から氏名変換
    Private Function NtoName(JJ As String) As String

        Dim cm As NpgsqlCommand
        Dim strSQL As String
        Dim rs As NpgsqlDataReader
        Dim Num As Integer

        If cn1.State = ConnectionState.Open = True Then
            cn1.Close()
        End If

        If Len(JJ) > 0 Then
            'マスタ登録チェック　削除されている可能性あり
            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM master_jyuban "
            strSQL = strSQL + "WHERE jno='" & JJ & "' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num = cm.ExecuteScalar
            cn1.Close()

            If Num <> 0 Then
                strSQL = ""
                strSQL = strSQL + "SELECT jname FROM master_jyuban "
                strSQL = strSQL + "WHERE jno='" & JJ & "' "
                cn1.Open()
                cm = New NpgsqlCommand
                cm.Connection = cn1
                cm.CommandText = strSQL
                rs = cm.ExecuteReader
                rs.Read()
                NtoName = rs("jname").ToString
                rs.Close()
                cn1.Close()
            Else
                NtoName = ""
            End If
        Else
            NtoName = ""
        End If

    End Function

    Private Sub View_C1FLG1(DE)

        Dim cm As NpgsqlCommand
        Dim strSQL As String
        Dim Num As Integer
        Dim rs As NpgsqlDataReader
        Dim i As Integer

        Try
            If cn1.State = ConnectionState.Open = True Then
                cn1.Close()
            End If

            '金型別対象データ検索
            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM work_real "
            strSQL = strSQL + "WHERE katano='" & DE & "' AND (pflg='0' OR pflg='1' OR pflg='9') AND sflg='0';"
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num = cm.ExecuteScalar
            cn1.Close()

            If Num = 0 Then
                'C1FlexGrid1.EndUpdate()
                C1FlexGrid1.Visible = False
                Exit Sub
            Else
                C1FlexGrid1.Rows.Count = Num + 1
                i = 1
            End If

            strSQL = ""
            strSQL = strSQL + "SELECT * FROM view_work_real "
            strSQL = strSQL + "WHERE katano='" & DE & "' AND (pflg='0' OR pflg='1' OR pflg='9') AND sflg='0' "
            strSQL = strSQL + "ORDER BY workno;"
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            rs = cm.ExecuteReader
            While (rs.Read())
                C1FlexGrid1(i, 0) = CStr(i)
                C1FlexGrid1(i, 1) = rs("katano").ToString

                C1FlexGrid1(i, 2) = rs("kouteino").ToString
                C1FlexGrid1(i, 3) = rs("jname").ToString

                '表示変換
                C1FlexGrid1(i, 5) = Strings.Mid(rs("starttime").ToString, 6)
                C1FlexGrid1(i, 6) = Strings.Mid(rs("endtime").ToString, 6)
                C1FlexGrid1(i, 7) = Strings.Mid(rs("worktime").ToString, 1, 8)

                If rs("cflg").ToString = "0" Then
                    C1FlexGrid1(i, 8) = ""
                ElseIf rs("cflg").ToString = "1" Then
                    C1FlexGrid1(i, 8) = "×"
                ElseIf rs("cflg").ToString = "9" Then
                    C1FlexGrid1(i, 8) = "△"
                Else

                End If

                i = i + 1
            End While
            cn1.Close()

            C1FlexGrid1.Select(0, 0, True)
            C1FlexGrid1.Visible = True

        Catch ex As Exception
            Call ErrLOG("View_C1FLG1", "")
        End Try

    End Sub

    Private Sub View_C1FLG2(KT As String, PN As String)

        Dim cm As NpgsqlCommand
        Dim strSQL As String
        Dim Num As Integer
        Dim Num1 As Integer
        Dim Num2 As Integer
        Dim rs As NpgsqlDataReader
        Dim i As Integer

        Dim tXT As String
        Dim eXT As String

        Try

            If cn1.State = ConnectionState.Open = True Then
                cn1.Close()
            End If
            C1FlexGrid2.BeginUpdate()

            '金型別対象データ検索
            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM work_real WHERE katano='" & KT & "' AND workno2='" & PN & "' AND sflg<>'1' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num1 = cm.ExecuteScalar
            cn1.Close()

            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM work_real_history WHERE katano='" & KT & "' AND pflg='1' AND workno2='" & PN & "' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num2 = cm.ExecuteScalar
            cn1.Close()

            Num = Num1 + Num2

            If Num = 0 Then
                C1FlexGrid2.EndUpdate()
                C1FlexGrid2.Visible = False
                Button3.Visible = False
                Exit Sub
            Else
                C1FlexGrid2.Rows.Count = Num + 1
                i = 1
                Button3.Visible = True
            End If

            '作業中を色分けする
            C1FlexGrid2.Styles.Add("White")
            C1FlexGrid2.Styles("White").BackColor = Color.White
            C1FlexGrid2.Styles.Add("SkyBlue")
            C1FlexGrid2.Styles("SkyBlue").BackColor = Color.SkyBlue

            strSQL = ""
            strSQL = strSQL + "SELECT * FROM "
            strSQL = strSQL + "(SELECT * FROM view_work_real WHERE sflg='0' UNION ALL SELECT * FROM view_work_real_history) AS WS "
            strSQL = strSQL + "WHERE katano='" & KT & "' AND workno2='" & PN & "'"
            strSQL = strSQL + "ORDER BY workno;"
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            rs = cm.ExecuteReader
            While (rs.Read())
                C1FlexGrid2(i, 0) = CStr(i)
                C1FlexGrid2(i, 1) = rs("katano").ToString
                C1FlexGrid2(i, 2) = rs("kouteino").ToString
                C1FlexGrid2(i, 3) = rs("jname").ToString

                '表示変換
                C1FlexGrid2(i, 4) = Strings.Mid(rs("starttime").ToString, 6)
                C1FlexGrid2(i, 5) = Strings.Mid(rs("endtime").ToString, 6)
                C1FlexGrid2(i, 6) = Strings.Mid(rs("worktime").ToString, 1, 8)

                If rs("cflg").ToString = "0" Then
                    C1FlexGrid2(i, 7) = ""
                ElseIf rs("cflg").ToString = "1" Then
                    C1FlexGrid2(i, 7) = "×"
                ElseIf rs("cflg").ToString = "9" Then
                    C1FlexGrid2(i, 7) = "△"
                Else
                End If

                '作業内容 修理内容
                If Len(rs("t1").ToString) = 0 And Len(rs("e1").ToString) = 0 Then
                    C1FlexGrid2(i, 8) = ""
                Else
                    If Len(rs("t1").ToString) > 0 Then
                        tXT = rs("t1").ToString & " " & rs("t2").ToString & " " & rs("t3").ToString & " " & rs("t4").ToString & " " & rs("t5").ToString & "   "
                    End If

                    If Len(rs("e1").ToString) > 0 Then
                        eXT = rs("e1").ToString & " " & rs("e2").ToString & " " & rs("e3").ToString & " " & rs("e4").ToString & " " & rs("e5").ToString
                    End If

#Disable Warning BC42104 ' 変数 'tXT' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                    If Len(tXT) = 0 Then
#Enable Warning BC42104 ' 変数 'tXT' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
#Disable Warning BC42104 ' 変数 'eXT' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                        C1FlexGrid2(i, 8) = "修理 : " & eXT
#Enable Warning BC42104 ' 変数 'eXT' は、値が割り当てられる前に使用されています。Null 参照の例外が実行時に発生する可能性があります。
                    Else
                        C1FlexGrid2(i, 8) = "作業 : " & tXT + "修理 : " & eXT
                    End If

                End If
                C1FlexGrid2(i, 9) = rs("workno").ToString

                '作業中
                If rs("pflg").ToString = "1" Then
                    C1FlexGrid2.Rows(i).Style = C1FlexGrid2.Styles("White")
                Else
                    C1FlexGrid2.Rows(i).Style = C1FlexGrid2.Styles("SkyBlue")
                End If

                i = i + 1
            End While
            cn1.Close()

            C1FlexGrid2.Select(0, 0, True)
            C1FlexGrid2.Visible = True
            C1FlexGrid2.EndUpdate()

        Catch ex As Exception
            Call ErrLOG("View_C1FLG2", "")
        End Try

    End Sub
