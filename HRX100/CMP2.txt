    '必要変数取得 INI.XML
    Private Sub Initialization()
        Try
            'INI.XMLから設定取得
            Dim MyPath As String = Application.StartupPath
            Dim iniFileName As String = "INI.xml"
            If MyPath.EndsWith("\") = False Then
                MyPath &= "\"
            End If
            iniFileName = MyPath & iniFileName

            '指定したXMLファイルの読み込み
            Dim xmlDoc As New System.Xml.XmlDocument
            xmlDoc.Load(iniFileName)
            Dim rootElement As System.Xml.XmlElement = xmlDoc.DocumentElement
            Dim node As System.Xml.XmlNodeList = rootElement.GetElementsByTagName("db_name")

            '指定したタグが存在するか？
            If node.Count > 0 Then
                DBN = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_port")
            If node.Count > 0 Then
                DBP = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("db_ip")
            If node.Count > 0 Then
                DIP = node.Item(0).InnerText
            End If

            'node = rootElement.GetElementsByTagName("usb_port")
            'If node.Count > 0 Then
            'CMP = node.Item(0).InnerText
            'End If

            node = rootElement.GetElementsByTagName("ppath")
            If node.Count > 0 Then
                PPH = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("acconect")
            If node.Count > 0 Then
                Accflg = node.Item(0).InnerText
            End If

            node = rootElement.GetElementsByTagName("koutei")
            If node.Count > 0 Then
                KNO = node.Item(0).InnerText
            End If

        Catch ex As Exception
            Call ErrLOG("Initialization", "")
            End
        End Try
    End Sub

    'DENSO WAVE Active USB-COM Port 取得
    Private Sub Init_DENSO_USB_COM_Port()

        Dim mcW32SerPort As New ManagementClass("Win32_SerialPort")
        Dim aSerialPort As ManagementObject
        Dim dCOM As String

        For Each aSerialPort In mcW32SerPort.GetInstances()
            Console.WriteLine(DirectCast(aSerialPort.GetPropertyValue("DeviceID"), String) & ":" & DirectCast(aSerialPort.GetPropertyValue("Caption"), String))
            'DENSO WAVE Active USB-COM Port 検索 AND Conneced検索
            dCOM = DirectCast(aSerialPort.GetPropertyValue("DeviceID"), String) & ":" & DirectCast(aSerialPort.GetPropertyValue("Caption"), String)

            If 0 <= dCOM.IndexOf("DENSO WAVE Active USB-COM Port") Then
                If 0 <= dCOM.IndexOf("Connected") Then
                    '接続ポート
                    CMP = Strings.Left(dCOM, 4)
                    Exit Sub
                Else

                End If
            Else
            End If
        Next
        CMP = ""

    End Sub

    Private Sub COM_Connect()

        Try
            If CMP <> "" Then
                '19200,N,8,1 通信設定
                With SerialPort1
                    .PortName = CMP
                    .BaudRate = 19200
                    .DataBits = 8
                    .Parity = Parity.None
                    .StopBits = StopBits.One
                    .Handshake = Handshake.None
                    .ReceivedBytesThreshold = 2
                    .ReadTimeout = 10000
                    .WriteTimeout = 10000

                    .DtrEnable = True
                    .RtsEnable = True
                End With

                If SerialPort1.IsOpen = False Then
                    SerialPort1.Open()
                    SerialPort1.DiscardInBuffer()
                    SerialPort1.DiscardOutBuffer()
                End If
            Else
                MessageBox.Show(" QRコードリーダーの接続が確認できません　", "警告", MessageBoxButtons.OK, MessageBoxIcon.Error)
            End If

        Catch ex As Exception
            Call ErrLOG("COM_Connect", "")
            MessageBox.Show(" ハンディターミナルが接続されていません??　", "ERROR", MessageBoxButtons.OK, MessageBoxIcon.Error)
        End Try

    End Sub
    Private Sub COM_DisConnect()
        If SerialPort1.IsOpen = True Then
            SerialPort1.Close()
        End If
    End Sub

    Private Sub SerialPort1_DataReceived(sender As Object, e As System.IO.Ports.SerialDataReceivedEventArgs) Handles SerialPort1.DataReceived

        If SerialPort1.IsOpen = False Then
            Return
        End If
        System.Threading.Thread.Sleep(500)
        'Clear
        blockbuf = ""
        Do
            Dim returnStr As String = SerialPort1.ReadExisting()
            blockbuf &= returnStr.ToString

            If SerialPort1.BytesToRead = 0 Then
                SerialPort1.DiscardInBuffer()
                SerialPort1.DiscardOutBuffer()
                Exit Do
            End If
        Loop

        '表示対応
        If Strings.Left(blockbuf, 2) = "DK" Then
            '手動モード確認
            If Panel1.Visible = True Then
                Panel1.Visible = False
                TextBox5.Text = ""

                TextBox1.Text = ""
                NPage = 0
                Tpage = 0
                PFNAME = ""
                Nkata = ""

                Button3.Enabled = False
                Button4.Enabled = False
                Button5.Enabled = False
                Button6.Enabled = False
                Button7.Enabled = False

                TextBox4.Text = KNO
                TextBox4.Enabled = False

                Label1.Visible = False
                TextBox1.Visible = False
            End If

            If Accflg = "0" Then
                '金型シート serverから検索
                Call DataMake01(blockbuf)
            Else
                'accdb への検索 2015/11/02
                Call DataMake11(blockbuf)
            End If
        ElseIf Strings.Left(blockbuf, 2) = "CC" Then
            'ページ変更他
            Call DataMake02(blockbuf)
        Else

        End If

    End Sub
    '金型シート情報検索
    Private Sub DataMake01(kata As String)

        Dim cm As NpgsqlCommand
        Dim strSQL As String
        Dim rs As NpgsqlDataReader
        Dim Num As Integer

        Try
            If Nkata = Strings.Mid(kata, 3, 5) Then
                Exit Sub
            Else
                '処理中_金型
                Nkata = Strings.Mid(kata, 3, 5)
            End If

            If cn1.State = ConnectionState.Open = True Then
                cn1.Close()
            End If

            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM master_die_pdf "
            strSQL = strSQL + "WHERE katano='" & Strings.Mid(kata, 3, 5) & "' AND koutei='" & KNO & "' "
            strSQL = strSQL + "AND dflg='0' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num = cm.ExecuteScalar
            cn1.Close()

            If Num = 0 Then
                'MessageBox.Show("", "警告", MessageBoxButtons.OK, MessageBoxIcon.Exclamation)
                PFNAME = ""

                TextBox1.Text = ""
                NPage = 0
                Tpage = 0
                PFNAME = ""
                Nkata = ""

                Button3.Enabled = False
                Button4.Enabled = False
                Button5.Enabled = False
                Button6.Enabled = False
                Button7.Enabled = False

                TextBox4.Text = KNO
                TextBox4.Enabled = False

                Label1.Visible = False
                TextBox1.Visible = False

                If Panel1.Visible = True Then
                    Panel1.Visible = False

                    TextBox5.Text = ""

                    TextBox1.Text = ""
                    NPage = 0
                    Tpage = 0
                    PFNAME = ""
                    Nkata = ""

                    Button3.Enabled = False
                    Button4.Enabled = False
                    Button5.Enabled = False
                    Button6.Enabled = False
                    Button7.Enabled = False

                    TextBox4.Text = KNO
                    TextBox4.Enabled = False

                    Label1.Visible = False
                    TextBox1.Visible = False

                End If

                '該当データなし
                Dim stCurrentDir As String = System.IO.Directory.GetCurrentDirectory()
                AxAcroPDF2.LoadFile(stCurrentDir & "\" & "default.pdf")
                AxAcroPDF2.setShowScrollbars(False)
                AxAcroPDF2.setShowToolbar(False)
                Label1.Visible = False

                Exit Sub
            End If

            strSQL = ""
            strSQL = strSQL + "SELECT * FROM master_die_pdf "
            strSQL = strSQL + "WHERE katano='" & Strings.Mid(kata, 3, 5) & "' AND koutei='" & KNO & "' "
            strSQL = strSQL + "AND dflg='0' "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            rs = cm.ExecuteReader
            rs.Read()

            '結合PDFの有無調査
            If rs("compage") <> 0 Then
                CFLG = True
                'ファイル名
                PFNAME = rs("compdf").ToString

                P1 = rs("tpage")

                If IsDBNull(rs("ytpage")) Then
                    P2 = 0
                Else
                    P2 = rs("ytpage")
                End If

                If IsDBNull(rs("rtpage")) Then
                    P3 = 0
                Else
                    P3 = rs("rtpage")
                End If

                If IsDBNull(rs("otpage")) Then
                    P4 = 0
                Else
                    P4 = rs("otpage")
                End If

                If IsDBNull(rs("prtpage")) Then
                    P5 = 0
                Else
                    P5 = rs("prtpage")
                End If

                'トータルページ数
                Tpage = rs("compage")
            Else
                CFLG = False
                PFNAME = rs("pfname").ToString
                Tpage = rs("tpage")

                P2 = 0
                P3 = 0
                P4 = 0
                P5 = 0

            End If

            rs.Close()
            cn1.Close()

            'local　ファイル有無確認
            '必要に応じてServerからCOPY
            If System.IO.File.Exists(PdfPath & "\" & PFNAME) = False Then
                If System.IO.File.Exists(PPH & "\" & PFNAME) = False Then
                    Dim stCurrentDir As String = System.IO.Directory.GetCurrentDirectory()
                    PFNAME = stCurrentDir & "\" & "default.pdf"
                Else
                    Call PDFcopy02(PFNAME)
                End If
            End If

            '表示
            TextBox1.Invoke(New dlgWriteTxt(AddressOf WriteTxt1), Strings.Mid(kata, 3, 5))
            NPage = 1

            Call UseLog(Strings.Mid(kata, 3, 5))

        Catch ex As Exception
            Call ErrLOG("DataMake01", "")
            If cn1.State = ConnectionState.Open = True Then
                cn1.Close()
            End If
        End Try

    End Sub

                    AxAcroPDF2.setZoom(110)
                ElseIf P2 + P3 + 1 = NPage Then
                    'デポ指示書 1ページ目
                    AxAcroPDF2.setZoom(61)
                ElseIf P2 + P3 + 1 < NPage And P2 + P3 + P1 + 1 > NPage Then
                    '複数ページ対応

                ElseIf P2 + P3 + P1 + 1 = NPage Then

                ElseIf P5 <> 0 Then
                    '作業指示書
                    AxAcroPDF2.setZoom(110)
                End If
            Else

            End If
        Else

        End If

    End Sub

    'シャットダウンするためのセキュリティ特権を有効にする
    Public Shared Sub AdjustToken()
        Const TOKEN_ADJUST_PRIVILEGES As Integer = &H20
        Const TOKEN_QUERY As Integer = &H8
        Const SE_PRIVILEGE_ENABLED As Integer = &H2
        Const SE_SHUTDOWN_NAME As String = "SeShutdownPrivilege"

        If Environment.OSVersion.Platform <> PlatformID.Win32NT Then
            Return
        End If

        Dim procHandle As IntPtr = GetCurrentProcess()

        'トークンを取得する
        Dim tokenHandle As IntPtr
        OpenProcessToken(procHandle, _
            TOKEN_ADJUST_PRIVILEGES Or TOKEN_QUERY, tokenHandle)
        'LUIDを取得する
        Dim tp As New TOKEN_PRIVILEGES()
        tp.Attributes = SE_PRIVILEGE_ENABLED
        tp.PrivilegeCount = 1
        LookupPrivilegeValue(Nothing, SE_SHUTDOWN_NAME, tp.Luid)
        '特権を有効にする
        AdjustTokenPrivileges(tokenHandle, False, tp, 0, IntPtr.Zero, IntPtr.Zero)

        '閉じる
        CloseHandle(tokenHandle)
    End Sub

    Private Sub Init_Combobox1()

        Dim cm As NpgsqlCommand
        Dim strSQL As String
        Dim rs As NpgsqlDataReader
        Dim Num As Integer

        Try
            If cn1.State = ConnectionState.Open = True Then
                cn1.Close()
            End If

            ComboBox1.Items.Clear()
            ComboBox1.Text = ""

            strSQL = ""
            strSQL = strSQL + "SELECT COUNT(*) FROM master_die "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            Num = cm.ExecuteScalar
            cn1.Close()

            If Num = 0 Then
                Exit Sub
            End If

            strSQL = ""
            strSQL = strSQL + "SELECT bname FROM master_die "
            strSQL = strSQL + "GROUP BY bname "
            strSQL = strSQL + "ORDER BY bname "
            cn1.Open()
            cm = New NpgsqlCommand
            cm.Connection = cn1
            cm.CommandText = strSQL
            rs = cm.ExecuteReader
            While (rs.Read())
                ComboBox1.Items.Add(rs("bname").ToString)
            End While
            cn1.Close()
            ComboBox1.Text = ""

            ComboBox2.Items.Clear()
            ComboBox2.Text = ""
            ComboBox3.Items.Clear()
            ComboBox3.Text = ""

        Catch ex As Exception
            Call ErrLOG("Init_Combobox1", "")
        End Try

    End Sub
